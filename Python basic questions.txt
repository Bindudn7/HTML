    l=[]
    for i in l1:

            l.extend(nestedloop(i))
        else:

CALP:
// Variables
variables
{
  msTimer speedTimer;   // Timer for periodic speed updates
  int vehicleSpeed = 0; // Current simulated speed
}

// Initialization
on start
{
  write("Speed Sensor ECU Simulation Started");
  setTimer(speedTimer, 100); // Trigger every 100 ms
}

// Periodic broadcast of speed
on timer speedTimer
{
  message 0x200 speedMsg;   // CAN message with ID 0x200
  speedMsg.len = 1;
  speedMsg.byte(0) = vehicleSpeed; // Put speed in first byte
  output(speedMsg);

  // Increment speed for simulation (looping 0–250 km/h)
  vehicleSpeed = (vehicleSpeed + 5) % 251;

  setTimer(speedTimer, 100); // Restart timer
}

// Respond to diagnostic request (UDS Read Data By Identifier)
on message 0x7E0
{
  if (this.byte(0) == 0x22) // Service ID: Read Data By Identifier
  {
    message diagResp;
    diagResp.id = 0x7E8;     // ECU response ID
    diagResp.len = 3;
    diagResp.byte(0) = 0x62; // Positive response (0x22 + 0x40)
    diagResp.byte(1) = 0xF1; // Example DID high byte
    diagResp.byte(2) = vehicleSpeed; // Return current speed
    output(diagResp);

    write("Responded to UDS Read DID with speed = %d km/h", vehicleSpeed);
  }
}


2.CAPL code for  When the user presses the R key, this script sends a UDS diagnostic request to reset the ECU 
// Send a diagnostic request (UDS service 0x10 - ECU Reset)
on key 'r'
{
  message diagReq;
  diagReq.id = 0x7E0;          // Tester-to-ECU ID
  diagReq.len = 2;
  diagReq.byte(0) = 0x10;      // Service ID: ECU Reset
  diagReq.byte(1) = 0x01;      // Sub-function: Hard Reset
  output(diagReq);
  write("Sent ECU Reset request");
}

3.This script sends a CAN message with ID 0x200 every second, simulating an ECU periodically broadcasting data.
// Send a CAN message every 1 second
variables
{
  msTimer sendTimer;
}

on start
{
  setTimer(sendTimer, 1000); // 1000 ms = 1 second
}

on timer sendTimer
{
  message 0x200 msg;   // Create message with ID 0x200
  msg.byte(0) = 0x55;  // First data byte
  output(msg);         // Transmit message
  setTimer(sendTimer, 1000); // Restart timer
}
